function weekPlan(base, ex, wk) {

  const steps = ex.comp ? wk-1 : Math.floor((wk-1)/2);

  return { w: +(Math.max(0, base + steps*ex.inc)).toFixed(1), lo: ex.lo, hi: ex.hi, s: ex.s };

}


function coachMsg(ex, plan, prevSets) {

  if (!plan) return null;

  if (!prevSets || !prevSets.length) return { type:'intro', msg:`Start at ${plan.w>0?plan.w+'kg':'BW / light'}. Hit ${plan.lo}–${plan.hi} reps on all ${ex.s} sets with solid form before we increase.` };

  const done = prevSets.filter(r=>r.d);

  if (!done.length) return { type:'intro', msg:`Start at ${plan.w}kg and aim for ${plan.lo}–${plan.hi} reps per set.` };

  const allMax = done.length >= ex.s && done.every(r=>parseInt(r.reps)>=plan.hi);

  const top = Math.max(...done.map(r=>parseFloat(r.kg)||0), 0);

  const avg = done.reduce((a,r)=>a+(parseInt(r.reps)||0),0)/done.length;

  if (allMax) return { type:'increase', msg:`You hit all ${plan.hi} reps last session — increase to ${plan.w}kg today. Aim for ${plan.lo} reps minimum every set.` };

  if (done.length < ex.s) return { type:'hold', msg:`Hold at ${top||plan.w}kg. Complete all ${ex.s} sets first — consistency before load.` };

  if (avg < plan.lo+1) return { type:'hold', msg:`Hold at ${top||plan.w}kg. Work up to ${plan.hi} reps per set, then we add weight.` };

  return { type:'progress', msg:`Good progress. Push for ${plan.hi} reps every set at ${top||plan.w}kg — hit that and next session we increase.` };

}

function renderHistory() {

  const hist = getHistory();

  if (!hist.length) return `<div class="empty"><div class="empty-icon">📋</div><div class="empty-txt">No sessions yet. Complete and save a workout.</div></div>`;


  const grouped = {};

  hist.forEach(h=>{

    const wk = getWeekLabel(h.date);

    if(!grouped[wk]) grouped[wk]=[];

    grouped[wk].push(h);

  });


  let html = `<div style="font-family:'Bebas Neue',sans-serif;font-size:26px;letter-spacing:2px;color:#fff;margin-bottom:4px;">Session History</div>

  <div style="font-size:11px;color:#333;margin-bottom:18px;">${hist.length} sessions · stored on device</div>`;


  Object.entries(grouped).forEach(([wk,entries])=>{

    html += `<div class="hist-week-lbl">${wk}</div>`;

    entries.forEach(h=>{

      const d = DAYS.find(d=>d.id===h.dayId); if(!d) return;

      const ts = Object.values(h.logs||{}).reduce((s,sets)=>s+sets.filter(r=>r.d).length,0);

      const allW = Object.values(h.logs||{}).flatMap(sets=>sets.map(r=>parseFloat(r.kg)).filter(Boolean));

      const top = allW.length?Math.max(...allW):null;

      html += `<div class="hist-entry" style="border-left:3px solid ${d.color};">

        <div>

          <div style="display:flex;align-items:center;">

            <span class="hist-focus" style="color:${d.color};">${d.focus}</span>

            <span class="hist-tag">${d.tag}</span>

          </div>

          <div class="hist-date">${fmtD(h.date)}</div>

        </div>

        <div style="text-align:right;">

          <div class="hist-sets">${h.completed}/${h.total} exercises</div>

          <div class="hist-detail">${ts} sets${top?` · top ${top}kg`:''}</div>

        </div>

      </div>`;

    });

  });


  if (hist.length>=2) html += renderChart(hist);

  return html;

}


function renderChart(hist) {

  const weeks={};

  hist.forEach(h=>{const w=weekStart(h.date);if(!weeks[w])weeks[w]=0;weeks[w]++;});

  const entries=Object.entries(weeks).slice(-8).map(([w,c])=>({label:w.slice(5),count:c}));

  const max=Math.max(...entries.map(e=>e.count),1);

  let bars='';

  entries.forEach(e=>{

    const h=Math.round((e.count/max)*38);

    bars+=`<div class="chart-bar-wrap">

      <div class="chart-bar-count">${e.count}</div>

      <div class="chart-bar" style="background:#E8FF47;opacity:${.2+(e.count/max)*.8};height:${h}px;"></div>

      <div class="chart-bar-date">${e.label}</div>

    </div>`;

  });

  return `<div class="chart-box"><div class="chart-lbl">WEEKLY SESSIONS</div><div class="chart-bars">${bars}</div></div>`;

}

function updateRow(i, field, val) {

  state.modalEx.rows[i][field] = val;

}


function toggleDone(i) {

  const r = state.modalEx.rows[i];

  r.d = !r.d;

  const d = DAYS[state.dayIdx];

  const btn = document.getElementById('done-'+i);

  const row = document.getElementById('srow-'+i);

  btn.style.borderColor = r.d ? d.color : '#2a2a2a';

  btn.style.background  = r.d ? d.color : 'transparent';

  btn.style.color        = r.d ? d.tc   : '#2a2a2a';

  btn.textContent        = r.d ? '✓' : '';

  row.style.opacity      = r.d ? '0.5' : '1';

  state.modalEx.rows[i].kg   = document.getElementById('kg-'+i).value;

  state.modalEx.rows[i].reps = document.getElementById('rp-'+i).value;

}


function saveModal() {

  const { exIdx, rows } = state.modalEx;

  // Flush input values before save

  rows.forEach((r,i)=>{

    r.kg   = document.getElementById('kg-'+i)?.value || r.kg;

    r.reps = document.getElementById('rp-'+i)?.value || r.reps;

  });

  const d = DAYS[state.dayIdx];

  const sessions = getSessions();

  const cur = sessions[d.id];

  const isToday = cur?.date===today();

  sessions[d.id] = { date:today(), logs:{ ...(isToday?cur.logs:{}), [exIdx]:rows } };

  saveSessions(sessions);

  startTimer(d.ex[exIdx].rest, d.color);

  closeModal();

  render();

}


function closeModal(e) {

  if (e && e.target !== document.getElementById('modal-overlay')) return;

  document.getElementById('modal-overlay').classList.remove('show');

  state.modalEx = null;

}

function startTimer(secs, color) {

  if (state.timerInterval) clearInterval(state.timerInterval);

  state.timer = { total:secs, left:secs, color };

  state.timerPaused = false;

  const box = document.getElementById('timer-box');

  if (box) box.classList.add('show');

  state.timerInterval = setInterval(tickTimer, 1000);

}


function tickTimer() {

  if (state.timerPaused || !state.timer) return;

  state.timer.left--;

  const arc = document.getElementById('timer-arc');

  const txt = document.getElementById('timer-txt');

  if (arc && txt) {

    const circ = 2*Math.PI*36;

    const pct = state.timer.left/state.timer.total;

    arc.setAttribute('stroke-dashoffset', circ*(1-pct));

    txt.textContent = fmtT(state.timer.left);

  }

  if (state.timer.left<=0) {

    clearInterval(state.timerInterval);

    state.timer = null;

    const box = document.getElementById('timer-box');

    if (box) box.classList.remove('show');

  }

}


function togglePause() {

  state.timerPaused = !state.timerPaused;

  const btn = document.querySelector('.timer-btns button');

  if (btn) btn.textContent = state.timerPaused?'RESUME':'PAUSE';

}


function skipTimer() {

  clearInterval(state.timerInterval);

  state.timer = null;

  const box = document.getElementById('timer-box');

  if (box) box.classList.remove('show');

}

// DATE UTILS

// ═══════════════════════════════════════════════════════════════

function weekStart(iso) {

  const d=new Date(iso); d.setDate(d.getDate()-d.getDay()+1);

  return d.toISOString().split('T')[0];

}

function getWeekLabel(iso) {

  const d=new Date(iso),now=new Date();

  const sw=new Date(now); sw.setDate(now.getDate()-now.getDay()+1); sw.setHours(0,0,0,0);

  const lw=new Date(sw); lw.setDate(sw.getDate()-7);

  if(d>=sw) return'This Week';

  if(d>=lw) return'Last Week';

  const ws=weekStart(iso);

  return`Week of ${new Date(ws).toLocaleDateString('en-GB',{day:'numeric',month:'short'})}`;

}
